\section*{题目1：模7计数器}

\textbf{题目}：设计一个模7计数器（0-6循环），写出VHDL代码和Testbench。

\textbf{分析}：
\begin{itemize}
  \item $N=7$，$k=\lceil\log_2 7\rceil = 3$位
  \item 清零条件：cnt = 6(110)
  \item 状态序列：000→001→010→011→100→101→110→000
\end{itemize}

\textbf{VHDL}：
\begin{lstlisting}[caption={模7计数器（0-6循环）}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mod7_counter is
  port (
    clk   : in  std_logic;
    rst_n : in  std_logic;
    q     : out std_logic_vector(2 downto 0)
  );
end mod7_counter;

architecture behavioral of mod7_counter is
  signal cnt : unsigned(2 downto 0);
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      cnt <= "000";
    elsif rising_edge(clk) then
      if cnt = 6 then
        cnt <= "000";
      else
        cnt <= cnt + 1;
      end if;
    end if;
  end process;

  q <= std_logic_vector(cnt);
end behavioral;
\end{lstlisting}

\section*{题目7：模10 BCD计数器带7段显示}

\textbf{题目}：设计模10 BCD计数器 + BCD-7段译码器。输出驱动7段数码管显示0-9。

\textbf{解}：
\begin{lstlisting}[caption={模10 BCD计数器+7段译码器}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity bcd_counter_7seg is
  port (
    clk    : in  std_logic;
    rst_n  : in  std_logic;
    seg    : out std_logic_vector(6 downto 0)  -- a~g，低有效
  );
end bcd_counter_7seg;

architecture behavioral of bcd_counter_7seg is
  signal cnt : unsigned(3 downto 0);
begin
  -- 模10 BCD计数器
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      cnt <= "0000";
    elsif rising_edge(clk) then
      if cnt = 9 then
        cnt <= "0000";
      else
        cnt <= cnt + 1;
      end if;
    end if;
  end process;

  -- BCD-7段译码器（共阳极：0亮1灭）
  process(cnt)
  begin
    case cnt is
      when "0000" => seg <= "1000000";  -- 0
      when "0001" => seg <= "1111001";  -- 1
      when "0010" => seg <= "0100100";  -- 2
      when "0011" => seg <= "0110000";  -- 3
      when "0100" => seg <= "0011001";  -- 4
      when "0101" => seg <= "0010010";  -- 5
      when "0110" => seg <= "0000010";  -- 6
      when "0111" => seg <= "1111000";  -- 7
      when "1000" => seg <= "0000000";  -- 8
      when "1001" => seg <= "0010000";  -- 9
      when others => seg <= "1111111";  -- 非法：全灭
    end case;
  end process;
end behavioral;
\end{lstlisting}

\section*{题目8：模60计数器（秒/分计数器）}

\textbf{题目}：设计模60计数器用于时钟应用。输出秒/分的十位和个位。

\textbf{解}：使用两个BCD计数器级联——个位模10（0-9），十位模6（0-5）。个位计到9时向下一个周期产生进位，十位加1。当十位=5且个位=9时，整体回0。

\begin{lstlisting}[caption={模60 BCD计数器}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mod60_counter is
  port (
    clk    : in  std_logic;
    rst_n  : in  std_logic;
    sec_lo : out std_logic_vector(3 downto 0);  -- 个位(0-9)
    sec_hi : out std_logic_vector(3 downto 0)   -- 十位(0-5)
  );
end mod60_counter;

architecture behavioral of mod60_counter is
  signal cnt_lo : unsigned(3 downto 0);  -- 个位
  signal cnt_hi : unsigned(3 downto 0);  -- 十位
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      cnt_lo <= "0000";
      cnt_hi <= "0000";
    elsif rising_edge(clk) then
      if cnt_hi = 5 and cnt_lo = 9 then
        cnt_lo <= "0000";
        cnt_hi <= "0000";
      elsif cnt_lo = 9 then
        cnt_lo <= "0000";
        cnt_hi <= cnt_hi + 1;
      else
        cnt_lo <= cnt_lo + 1;
      end if;
    end if;
  end process;

  sec_lo <= std_logic_vector(cnt_lo);
  sec_hi <= std_logic_vector(cnt_hi);
end behavioral;
\end{lstlisting}

\section*{题目12：给定VHDL推断模值}

\textbf{题目}：分析以下VHDL代码所描述的计数器模值。
\begin{lstlisting}[caption={推断模值——完整代码}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity mystery_counter is
  port (
    clk   : in  std_logic;
    rst_n : in  std_logic;
    q     : out std_logic_vector(3 downto 0)
  );
end mystery_counter;

architecture behavioral of mystery_counter is
  signal cnt : unsigned(3 downto 0);
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      cnt <= "0000";
    elsif rising_edge(clk) then
      if cnt = "1010" then
        cnt <= "0000";
      else
        cnt <= cnt + 1;
      end if;
    end if;
  end process;

  q <= std_logic_vector(cnt);
end behavioral;
\end{lstlisting}
求模值N。

\textbf{解}：清零条件是cnt=10(1010)，所以N=10+1=11？不对！
清零条件=cnt=1010=10，此时cnt还未计数到11就被清零了。
模值=清零条件值+1=10+1=11？再确认：cnt从0计到10清零，所以N=11。

\textbf{关键公式}：模值 = 清零比较值 + 1

\section*{题目13：分析计数器最大工作频率}

\textbf{题目}：4位行波进位加法器的进位链延迟为每个FA=2ns，D触发器$t_{setup}=1$ns，$t_{cko}=1$ns。求计数器的最大工作频率。

\textbf{分析}：
关键路径延迟 = $t_{cko}$ + 进位链延迟(4×2=8ns) + $t_{setup}$ = 1+8+1 = 10ns
$f_{max} = 1/10\text{ns} = 100\text{MHz}$

\section*{题目14：多个计数器级联的分析}

\textbf{题目}：模8计数器 + 模5计数器级联。总模值是多少？

\textbf{解}：模8每8拍产生一次进位，模5每收到进位+1。
总模值 = 8×5 = 40。

\textbf{通用公式}：级联总模值 = $M_1 \times M_2 \times \dots \times M_k$

\section*{题目16-20：综合应用题}

\subsection*{题目16：数字钟核心}
设计时分秒计数器（模60秒+模60分+模24时）。

\textbf{解}：三个计数器级联——秒(模60)→分(模60)→时(模24)。
\begin{lstlisting}[caption={数字钟核心——时分秒计数器}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity digital_clock is
  port (
    clk     : in  std_logic;
    rst_n   : in  std_logic;
    sec_lo  : out std_logic_vector(3 downto 0);
    sec_hi  : out std_logic_vector(3 downto 0);
    min_lo  : out std_logic_vector(3 downto 0);
    min_hi  : out std_logic_vector(3 downto 0);
    hour_lo : out std_logic_vector(3 downto 0);
    hour_hi : out std_logic_vector(3 downto 0)
  );
end digital_clock;

architecture behavioral of digital_clock is
  signal s_lo, s_hi : unsigned(3 downto 0);  -- 秒
  signal m_lo, m_hi : unsigned(3 downto 0);  -- 分
  signal h_lo, h_hi : unsigned(3 downto 0);  -- 时
  signal s_carry : std_logic;  -- 秒→分进位
  signal m_carry : std_logic;  -- 分→时进位
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      s_lo <= "0000"; s_hi <= "0000";
      m_lo <= "0000"; m_hi <= "0000";
      h_lo <= "0000"; h_hi <= "0000";
      s_carry <= '0'; m_carry <= '0';
    elsif rising_edge(clk) then
      -- 秒计数器（模60）
      if s_hi = 5 and s_lo = 9 then
        s_lo <= "0000"; s_hi <= "0000";
        s_carry <= '1';
      elsif s_lo = 9 then
        s_lo <= "0000"; s_hi <= s_hi + 1;
        s_carry <= '0';
      else
        s_lo <= s_lo + 1;
        s_carry <= '0';
      end if;
      -- 分计数器（模60，秒进位触发）
      if s_carry = '1' then
        if m_hi = 5 and m_lo = 9 then
          m_lo <= "0000"; m_hi <= "0000";
          m_carry <= '1';
        elsif m_lo = 9 then
          m_lo <= "0000"; m_hi <= m_hi + 1;
          m_carry <= '0';
        else
          m_lo <= m_lo + 1;
          m_carry <= '0';
        end if;
      else
        m_carry <= '0';
      end if;
      -- 时计数器（模24，分进位触发）
      if m_carry = '1' then
        if h_hi = 2 and h_lo = 3 then
          h_lo <= "0000"; h_hi <= "0000";
        elsif h_lo = 9 then
          h_lo <= "0000"; h_hi <= h_hi + 1;
        else
          h_lo <= h_lo + 1;
        end if;
      end if;
    end if;
  end process;

  sec_lo  <= std_logic_vector(s_lo);
  sec_hi  <= std_logic_vector(s_hi);
  min_lo  <= std_logic_vector(m_lo);
  min_hi  <= std_logic_vector(m_hi);
  hour_lo <= std_logic_vector(h_lo);
  hour_hi <= std_logic_vector(h_hi);
end behavioral;
\end{lstlisting}

\subsection*{题目18：带预置位的倒计时计数器}
从预置值倒计时到0，到0后重新加载预置值。

\textbf{解}：
\begin{lstlisting}[caption={带预置位的倒计时计数器}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity countdown_preset is
  port (
    clk     : in  std_logic;
    rst_n   : in  std_logic;
    preset  : in  std_logic_vector(3 downto 0);
    q       : out std_logic_vector(3 downto 0);
    zero    : out std_logic  -- 倒计时到0标志
  );
end countdown_preset;

architecture behavioral of countdown_preset is
  signal cnt : unsigned(3 downto 0);
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      cnt <= unsigned(preset);
    elsif rising_edge(clk) then
      if cnt = 0 then
        cnt <= unsigned(preset);  -- 到0后重新加载
      else
        cnt <= cnt - 1;
      end if;
    end if;
  end process;

  q    <= std_logic_vector(cnt);
  zero <= '1' when cnt = 0 else '0';
end behavioral;
\end{lstlisting}

\subsection*{题目19：两相正交计数器}
输出两个相位差90度的方波（格雷码计数器的一种应用）。

\textbf{解}：使用2位格雷码计数器产生正交输出。序列：00→01→11→10→00。
\begin{lstlisting}[caption={两相正交计数器（格雷码）}]
library ieee;
use ieee.std_logic_1164.all;

entity quadrature_counter is
  port (
    clk   : in  std_logic;
    rst_n : in  std_logic;
    phase_a : out std_logic;  -- 相位A
    phase_b : out std_logic   -- 相位B（滞后90度）
  );
end quadrature_counter;

architecture behavioral of quadrature_counter is
  signal gray : std_logic_vector(1 downto 0);
begin
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      gray <= "00";
    elsif rising_edge(clk) then
      case gray is
        when "00"   => gray <= "01";
        when "01"   => gray <= "11";
        when "11"   => gray <= "10";
        when "10"   => gray <= "00";
        when others => gray <= "00";
      end case;
    end if;
  end process;

  phase_a <= gray(0);
  phase_b <= gray(1);
end behavioral;
\end{lstlisting}

\subsection*{题目20：用计数器实现序列检测}
计数器状态作为地址，ROM存放序列，输出+检测逻辑。

\textbf{解}：计数器遍历状态作为ROM地址，ROM预先存放待检测序列的各bit，逐拍与输入比较。
\begin{lstlisting}[caption={计数器+ROM实现序列检测}]
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity counter_seq_detector is
  port (
    clk       : in  std_logic;
    rst_n     : in  std_logic;
    data_in   : in  std_logic;
    detected  : out std_logic
  );
end counter_seq_detector;

architecture behavioral of counter_seq_detector is
  signal addr : unsigned(1 downto 0);  -- 0-3，遍历4个序列位
  type rom_type is array (0 to 3) of std_logic;
  -- ROM存放目标序列"1011"（地址0→bit3, 1→bit2, 2→bit1, 3→bit0）
  constant pattern : rom_type := ('1', '0', '1', '1');
  signal match : std_logic_vector(3 downto 0);  -- 逐位匹配标志
begin
  -- 地址计数器：0→1→2→3→0循环
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      addr <= "00";
    elsif rising_edge(clk) then
      if addr = 3 then
        addr <= "00";
      else
        addr <= addr + 1;
      end if;
    end if;
  end process;

  -- 比对逻辑
  process(clk, rst_n)
  begin
    if rst_n = '0' then
      match <= (others => '0');
    elsif rising_edge(clk) then
      if data_in = pattern(to_integer(addr)) then
        match(to_integer(addr)) <= '1';
      else
        match(to_integer(addr)) <= '0';
      end if;
    end if;
  end process;

  -- 全部4位匹配→检测到
  detected <= '1' when match = "1111" else '0';
end behavioral;
\end{lstlisting}

\begin{examtip}[计数器类题目的统一解法]
\textbf{第一步：}确定模值N
\textbf{第二步：}$k = \lceil \log_2 N \rceil$，确定位宽
\textbf{第三步：}清零条件 = N-1 的二进制
\textbf{第四步：}写VHDL模板（if cnt=N-1 then cnt<=0 else cnt<=cnt+1）
\textbf{第五步：}如果有级联，低位进位接高位时钟使能

\textbf{关键公式}：
\begin{itemize}
  \item 模值 = 清零比较值 + 1
  \item 总模值（级联）= $M_1 \times M_2$
  \item $2^N$模值不需要显式清零
\end{itemize}
\end{examtip}
